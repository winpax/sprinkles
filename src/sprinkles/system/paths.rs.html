1use std::path::PathBuf;
2
3#[derive(Debug, Copy, Clone)]
4#[non_exhaustive]
5#[allow(clippy::enum_variant_names, dead_code)]
6/// This is a non-exhaustive list CSIDLs for Windows defined paths
7pub enum Paths {
8    /// System wide application data
9    ///
10    /// `CommonAppData` on Windows
11    CommonAppData,
12    /// Persistent application data for the current user
13    ///
14    /// `AppData` on Windows
15    AppData,
16    /// Non-persistent application data for the current user
17    ///
18    /// `LocalAppData` on Windows
19    LocalAppData,
20}
21
22#[cfg(windows)]
23impl Paths {
24    pub(crate) fn as_csidl(self) -> u32 {
25        use windows::Win32::UI::Shell::{CSIDL_APPDATA, CSIDL_COMMON_APPDATA, CSIDL_LOCAL_APPDATA};
26
27        match self {
28            Paths::CommonAppData => CSIDL_COMMON_APPDATA,
29            Paths::AppData => CSIDL_APPDATA,
30            Paths::LocalAppData => CSIDL_LOCAL_APPDATA,
31        }
32    }
33
34    pub fn into_path(self) -> Option<PathBuf> {
35        use std::{ffi::OsString, os::windows::ffi::OsStringExt};
36
37        use windows::Win32::{Foundation::MAX_PATH, UI::Shell::SHGetSpecialFolderPathW};
38
39        let mut buf = [0u16; MAX_PATH as usize];
40        let success = unsafe {
41            #[allow(clippy::cast_possible_wrap)]
42            SHGetSpecialFolderPathW(None, &mut buf, self.as_csidl() as i32, true).as_bool()
43        };
44
45        if success {
46            let string = OsString::from_wide(&buf);
47
48            let utf8_string = string.to_string_lossy();
49            let trimmed = utf8_string.trim_end_matches('\0');
50
51            Some(PathBuf::from(trimmed))
52        } else {
53            None
54        }
55    }
56}
57
58#[cfg(not(windows))]
59impl Paths {
60    #[allow(clippy::unused_self)]
61    pub fn into_path(self) -> Option<PathBuf> {
62        use std::env;
63
64        match self {
65            Paths::CommonAppData => Some(PathBuf::from("/usr/share")),
66            Paths::AppData => env::var_os("XDG_DATA_HOME").map(PathBuf::from).or_else(|| {
67                env::var_os("HOME").map(|home| PathBuf::from(home).join(".local/share"))
68            }),
69            Paths::LocalAppData => env::var_os("XDG_CACHE_HOME")
70                .map(PathBuf::from)
71                .or_else(|| env::var_os("HOME").map(|home| PathBuf::from(home).join(".cache"))),
72        }
73    }
74}
