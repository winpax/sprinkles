152impl Scoop {
153    /// Converts the config path into the [`Scoop`] struct
154    ///
155    /// # Errors
156    /// - The file was not valid UTF-8
157    /// - The read file was did not match the expected structure
158    pub fn load() -> std::io::Result<Self> {
159        let config_path = Self::get_path();
160
161        let config = std::fs::read_to_string(config_path)?;
162
163        let config: Self = serde_json::from_str(&config)?;
164
165        Ok(config)
166    }
167
168    #[must_use]
169    /// Gets the scoop config path
170    ///
171    /// # Panics
172    /// - The config directory does not exist
173    pub fn get_path() -> PathBuf {
174        let config_dir = crate::env::paths::config_dir();
175
176        let path = config_dir
177            .expect("Could not find config directory")
178            .join("scoop")
179            .join("config.json");
180
181        assert!(path.exists(), "Could not find config file");
182
183        path
184    }
185
186    /// Update the last time the scoop was updated
187    pub fn update_last_update_time(&mut self) {
188        let date_time = chrono::Local::now().to_rfc3339_opts(chrono::SecondsFormat::Micros, false);
189
190        self.last_update = Some(date_time.to_string());
191    }
192
193    /// Save the modified scoop config
194    ///
195    /// # Errors
196    /// - The struct could not be serialized to JSON
197    /// - The file could not be written
198    pub fn save(&self) -> std::io::Result<()> {
199        let config_path = Self::get_path();
200
201        let config = serde_json::to_string_pretty(self)?;
202
203        std::fs::write(config_path, config)?;
204
205        Ok(())
206    }
207
208    /// Make the config strict
209    ///
210    /// This will remove all fields that are not in the config struct
211    pub fn make_strict(&mut self) {
212        self.other = Map::new();
213    }
214
215    /// Convert the config to a JSON object
216    ///
217    /// # Errors
218    /// - The config could not be converted to a JSON object
219    pub fn to_object(&self) -> serde_json::Result<Value> {
220        serde_json::to_value(self)
221    }
222
223    /// Convert the JSON object to [`Scoop`] config
224    ///
225    /// # Errors
226    /// - The JSON object could not be deserialized to a [`Scoop`] config
227    pub fn from_object(object: Value) -> serde_json::Result<Self> {
228        serde_json::from_value(object)
229    }
230}

238    #[test]
239    fn test_isolated_path() {
240        let path = IsolatedPath::Path("NOT_SCOOP_PATH".into());
241        let serialized = serde_json::to_string(&path).unwrap();
242        let deserialized: IsolatedPath = serde_json::from_str(serialized.as_str()).unwrap();
243
244        assert_eq!(path, deserialized);
245
246        let path = "true";
247        let deserialized: IsolatedPath = serde_json::from_str(path).unwrap();
248
249        assert_eq!(IsolatedPath::Bool(true), deserialized);
250
251        let path = "false";
252        let deserialized: IsolatedPath = serde_json::from_str(path).unwrap();
253
254        assert_eq!(IsolatedPath::Bool(false), deserialized);
255    }
