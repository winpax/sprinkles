1use sxd_document::parser;
2use sxd_xpath::{evaluate_xpath, Value};
3
4use crate::hash::substitutions::{Substitute, SubstitutionMap};
5
6#[derive(Debug, thiserror::Error)]
7pub enum XMLError {
8    #[error("XML error: {0}")]
9    Xml(#[from] quick_xml::DeError),
10    #[error("XML Parsing: {0}")]
11    SXDXml(#[from] sxd_document::parser::Error),
12    #[error("XML XPath: {0}")]
13    XPath(#[from] sxd_xpath::Error),
14    #[error("Hash not found")]
15    NotFound,
16    #[error("Invalid value")]
17    InvalidValue,
18}
19
20pub fn parse_xml(
21    source: impl AsRef<str>,
22    substitutions: &SubstitutionMap,
23    xpath: impl AsRef<str>,
24) -> Result<String, XMLError> {
25    let mut xpath = xpath.as_ref().to_string();
26    xpath.substitute(substitutions, false);
27
28    let pkg = parser::parse(source.as_ref())?;
29    let doc = pkg.as_document();
30
31    let value = evaluate_xpath(&doc, xpath.as_ref())?;
32
33    let hash = match value {
34        Value::Nodeset(nodes) => {
35            let node = nodes.iter().last().ok_or(XMLError::NotFound)?;
36
37            node.string_value()
38        }
39        Value::String(text) => text,
40        _ => return Err(XMLError::InvalidValue),
41    };
42
43    Ok(hash)
44}
