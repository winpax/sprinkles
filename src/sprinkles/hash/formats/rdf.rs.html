9#[derive(Debug, thiserror::Error)]
10pub enum RDFError {
11    #[error("Hash could not be found in file")]
12    NotFound,
13}

15pub fn parse_xml(input: impl AsRef<[u8]>, file_name: impl AsRef<str>) -> Result<String, RDFError> {
16    use quick_xml::{events::Event, reader::Reader};
17
18    let mut reader = Reader::from_reader(input.as_ref());
19    reader.config_mut().trim_text(true);
20
21    let mut buf = vec![];
22
23    let mut about_tag = None;
24    let mut hash_tag = None;
25
26    let mut hash = None;
27
28    loop {
29        match reader.read_event_into(&mut buf) {
30            Err(e) => panic!("Error at position {}: {:?}", reader.buffer_position(), e),
31            Ok(Event::Eof) => break,
32
33            Ok(Event::Start(e)) => {
34                if about_tag.is_some() {
35                    if e.name() == QName(b"digest:sha256") {
36                        assert!(
37                            hash_tag.is_none(),
38                            "woops something wasnt cleaned up properly"
39                        );
40
41                        std::mem::swap(&mut about_tag, &mut hash_tag);
42                    }
43                } else {
44                    for attribute in e.attributes().flatten() {
45                        let file_name = file_name.as_ref().to_string();
46
47                        if attribute.key == QName(b"rdf:about")
48                            && attribute.value.as_ref() == file_name.as_bytes()
49                        {
50                            about_tag = Some(file_name);
51                        }
52                    }
53                }
54            }
55
56            Ok(Event::Text(e)) => {
57                if hash_tag.is_some() {
58                    hash = Some(e.unescape().unwrap().to_string());
59                    break;
60                }
61            }
62
63            _ => (),
64        }
65    }
66
67    hash.ok_or(RDFError::NotFound)
68}
