34fn query_jp<'a>(

37    substitutions: &SubstitutionMap,

41        jp.to_string().into_substituted(substitutions, regex_escape)
42    };
43
44    let path = JsonPath::parse(&jp)?;
45
46    Ok(path.query(json))
